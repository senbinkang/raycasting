## 1. P0-1：满血不捡医疗包（Player.js）

`pickupItems()` 中加一行判断：

```javascript
pickupItems() {
    if (!this.spriteManager) return
    for (let s of this.spriteManager.sprites) {
        if (!s.alive || s.type !== 'item') continue
        let dx = this.position.x - s.x
        let dy = this.position.y - s.y
        let pickRadius = 0.5
        if (dx * dx + dy * dy < pickRadius * pickRadius) {
            if (this.hp >= this.maxHp) continue   // 满血不捡
            s.alive = false
            this.hp = Math.min(this.maxHp, this.hp + 30)
        }
    }
}
```

## 4. P1-4：医疗包排除敌人周围 1 格（game/GameScene.js）

`_placeSprites()` 中放物品之前，检查该位置 1 格内是否有已放置的敌人：

```javascript
_placeSprites() {
    const empties = this._findEmptyCells()
    for (let i = empties.length - 1; i > 0; i--) {
        let j = Math.floor(Math.random() * (i + 1))
        ;[empties[i], empties[j]] = [empties[j], empties[i]]
    }
    let enemyCount = 0
    let itemCount = 0
    let enemyPositions = []   // 记录敌人位置
    const playerStartX = Math.floor(this.bg.columns / 2) + 0.5
    const playerStartY = Math.floor(this.bg.lines / 2) + 0.5

    for (let [cx, cy] of empties) {
        let dx = cx + 0.5 - playerStartX
        let dy = cy + 0.5 - playerStartY
        let dist = Math.sqrt(dx * dx + dy * dy)
        if (dist < 1.5) continue

        if (enemyCount < 4) {
            this._spawnEnemy(cx, cy)
            enemyPositions.push([cx, cy])
            enemyCount++
        } else if (itemCount < 5) {
            // 排除敌人周围 1 格
            let tooClose = false
            for (let [ex, ey] of enemyPositions) {
                if (Math.abs(cx - ex) <= 1 && Math.abs(cy - ey) <= 1) {
                    tooClose = true; break
                }
            }
            if (tooClose) continue
            this.spriteManager.add(new Sprite(cx + 0.5, cy + 0.5, 202, {
                type: 'item', isPickable: true
            }))
            itemCount++
        }
        if (enemyCount >= 4 && itemCount >= 5) break
    }
}
```

## 5. P1-5：每波重生放医疗包（game/GameScene.js）

`_respawnEnemies()` 生成敌人后，顺便在剩余空地中随机放 3 个医疗包：

```javascript
_respawnEnemies() {
    this.wave++
    if (this.wave >= 5) {
        this.state = 'win'
        this.overlay.show()
        this.overlay.drawWin()
        return
    }
    let empties = this._findEmptyCells()
    let px = this.player.position.x, py = this.player.position.y
    empties = empties.filter(([cx, cy]) => {
        let dx = cx + 0.5 - px, dy = cy + 0.5 - py
        return Math.sqrt(dx * dx + dy * dy) > 2
    })
    for (let i = empties.length - 1; i > 0; i--) {
        let j = Math.floor(Math.random() * (i + 1))
        ;[empties[i], empties[j]] = [empties[j], empties[i]]
    }
    let count = Math.min(4 + this.wave, empties.length)
    let enemyPositions = []
    for (let i = 0; i < count; i++) {
        let [cx, cy] = empties[i]
        this._spawnEnemy(cx, cy)
        enemyPositions.push([cx, cy])
    }
    // 放 3 个医疗包，排除敌人周围 1 格
    let itemCount = 0
    for (let i = count; i < empties.length && itemCount < 3; i++) {
        let [cx, cy] = empties[i]
        let tooClose = false
        for (let [ex, ey] of enemyPositions) {
            if (Math.abs(cx - ex) <= 1 && Math.abs(cy - ey) <= 1) {
                tooClose = true; break
            }
        }
        if (tooClose) continue
        this.spriteManager.add(new Sprite(cx + 0.5, cy + 0.5, 202, {
            type: 'item', isPickable: true
        }))
        itemCount++
    }
}
```
