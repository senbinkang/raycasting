## 二、逐项方案

### 2.1 玩家不被敌人阻挡（`Player.js`）

`_collidesBlockingSprite()` 中加一行：排除 `type === 'enemy'` 的精灵。

```javascript
_collidesBlockingSprite(x, y) {
    if (!this.spriteManager || ...) return false
    for (let s of this.spriteManager.sprites) {
        if (!s.alive) continue
        if (!s.isBlocking) continue
        if (s.type === 'enemy') continue   // 玩家可穿过敌人
        let dx = x - s.x
        let dy = y - s.y
        let minDist = this.collisionRadius + (s.radius || 0.2)
        if (dx * dx + dy * dy < minDist * minDist) return true
    }
    return false
}
```

### 2.2 强化墙面碰撞（`Player.js`）

`tryMove()` 中 X 轴检测增加到 5 个采样点（center, center±0.4r, center±0.8r），覆盖整个碰撞圆：

```javascript
tryMove(dx, dy, step) {
    let len = Math.sqrt(dx * dx + dy * dy)
    if (len < 0.0001) return
    dx = dx / len
    dy = dy / len

    let r = this.collisionRadius

    let newX = this.position.x + dx * step
    let newY = this.position.y + dy * step
    let checkX = newX + Math.sign(dx) * r
    let checkY = newY + Math.sign(dy) * r

    // 5 点采样覆盖整个碰撞圆
    let canMoveX = true
    for (let offset of [0, 0.4, -0.4, 0.8, -0.8]) {
        let sy = this.position.y + offset * r
        if (!this.cellIsEmpty(checkX, sy)) { canMoveX = false; break }
    }

    let canMoveY = true
    for (let offset of [0, 0.4, -0.4, 0.8, -0.8]) {
        let sx = this.position.x + offset * r
        if (!this.cellIsEmpty(sx, checkY)) { canMoveY = false; break }
    }

    if (canMoveX) this.position.x = newX
    if (canMoveY) this.position.y = newY
}
```

### 2.3 敌人全灭后重新生成（`GameScene.js`）

修改 `update()` 中的通关检测：

```javascript
// 替换原有的 win 检测
let enemies = this.spriteManager.sprites.filter(s => s.type === 'enemy')
let aliveEnemies = enemies.filter(s => s.alive)
if (enemies.length > 0 && aliveEnemies.length === 0) {
    this._respawnEnemies()
}
```

新增 `_respawnEnemies()` 方法：

```javascript
_respawnEnemies() {
    let empties = this._findEmptyCells()
    // 排除玩家附近 2 格
    let px = this.player.position.x, py = this.player.position.y
    empties = empties.filter(([cx, cy]) => {
        let dx = cx + 0.5 - px, dy = cy + 0.5 - py
        return Math.sqrt(dx * dx + dy * dy) > 2
    })
    // 随机洗牌取前 4 个
    for (let i = empties.length - 1; i > 0; i--) {
        let j = Math.floor(Math.random() * (i + 1));
        [empties[i], empties[j]] = [empties[j], empties[i]]
    }
    let count = Math.min(4, empties.length)
    for (let i = 0; i < count; i++) {
        let [cx, cy] = empties[i]
        this.spriteManager.add(new Sprite(cx + 0.5, cy + 0.5, 201, {
            type: 'enemy', speed: 1.2, hp: 100
        }))
    }
}
```

### 2.4 玩家初始位置 + 朝向（`Player.js`）

修改构造函数中的初始值和 `_findStartPosition()`：

```javascript
// 初始位置：地图中央
this.position = new Vec(
    Math.floor(this.columns / 2) + 0.5,
    Math.floor(this.lines / 2) + 0.5
)

// 初始朝向：朝上（-Y）
this.dirX = 0.0
this.dirY = -1.0
this.planeX = 0.66
this.planeY = 0.0
```

`_findStartPosition()` 候选列表改为以中央为起点向外扩展：

```javascript
_findStartPosition() {
    let cx = Math.floor(this.columns / 2)
    let cy = Math.floor(this.lines / 2)
    if (this.worldMap[cy][cx] === 0) return
    // 从中央螺旋向外找
    for (let r = 1; r < Math.max(this.columns, this.lines); r++) {
        for (let dx = -r; dx <= r; dx++) {
            for (let dy = -r; dy <= r; dy++) {
                let mx = cx + dx, my = cy + dy
                if (mx >= 0 && my >= 0 && mx < this.columns && my < this.lines) {
                    if (this.worldMap[my][mx] === 0) {
                        this.position = new Vec(mx + 0.5, my + 0.5)
                        return
                    }
                }
            }
        }
    }
}
```
